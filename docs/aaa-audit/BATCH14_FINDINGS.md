### B14-C-2 Fix: Add tidal_zone_label uint8 to detect_tidal_zones()

```python
# coastline.py:detect_tidal_zones() — after computing tidal_f32
# 5-zone classification: subtidal=0, intertidal=1, splash=2, spray=3, supralittoral=4
high = sea_level_m + tidal_range_m * 0.5
low = sea_level_m - tidal_range_m * 0.5
splash_top = high + tidal_range_m
spray_top = high + 3.0 * tidal_range_m
label = np.zeros_like(h, dtype=np.uint8)             # 0 = subtidal
label[h >= low] = 1                                   # intertidal
label[h >= high] = 2                                  # splash
label[h >= splash_top] = 3                            # spray
label[h >= spray_top] = 4                             # supralittoral
stack.set("tidal_zone_label", label, "coastline")
```
