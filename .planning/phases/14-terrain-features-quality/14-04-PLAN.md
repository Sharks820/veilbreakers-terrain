### BUG-94 fix in terrain_wind_erosion.py

Read the current apply_wind_erosion function (lines 82-119). The two buggy lines are:


Replace the entire shift-and-blend block with a continuous gradient approach. The new
implementation:

1. Builds a per-cell directional gradient along the wind vector using np.gradient.
   `grad_y, grad_x = np.gradient(h)` gives (dh/dy, dh/dx) in array coordinates.
2. Projects onto the wind unit vector:
   `wind_proj = dx * grad_x + dy * grad_y`
3. Constructs the "upwind" and "downwind" neighbour values using bilinear interpolation at
   fractional pixel offsets instead of integer-snapped shifts.
   Use `scipy.ndimage.map_coordinates` with fractional offsets of +-1.0 in the wind direction.
   `coords_up_r = row_coords - dy`, `coords_up_c = col_coords - dx`
   `coords_dn_r = row_coords + dy`, `coords_dn_c = col_coords + dx`
   Clamp coords to [0, H-1] and [0, W-1] (mode='nearest').
4. Blend: `blended = 0.5 * h + 0.3 * up + 0.2 * down` (same weights as before).
5. `delta = (blended - h) * intensity`

Exact replacement — delete lines 100-112 (the row/col_shift block through `blended = ...`)
and replace with:

Keep the coarse-grid + bilinear-upsample approach (it's fast), but make the coarse grid
seed depend on world coordinates, not the tile-level seed alone:

```python
    rng_base = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    h_arr, w_arr = shape
    gh = max(2, int(math.ceil(h_arr / max(scale_cells, 1.0))) + 2)
    gw = max(2, int(math.ceil(w_arr / max(scale_cells, 1.0))) + 2)

    # Per-cell world-space seed: each coarse-grid cell gets a unique seed derived
    # from its world position (world_row, world_col) so adjacent tiles agree at the seam.
    # World coords for coarse grid cell (gi, gj):
    #   world_row = world_row_offset + int(gi * scale_cells)
    #   world_col = world_col_offset + int(gj * scale_cells)
    gi_arr = np.arange(gh, dtype=np.int64)
    gj_arr = np.arange(gw, dtype=np.int64)
    world_rows = (world_row_offset + (gi_arr * int(scale_cells))).reshape(-1, 1)  # (gh, 1)
    world_cols = (world_col_offset + (gj_arr * int(scale_cells))).reshape(1, -1)  # (1, gw)
    # XOR hash combining tile seed + world position
    cell_seeds = (
        int(seed)
        ^ (world_rows.astype(np.int64) * 73856093)
        ^ (world_cols.astype(np.int64) * 19349663)
    ).astype(np.int64) & 0xFFFFFFFF

    # Sample each coarse cell individually using its world seed
    grid = np.empty((gh, gw), dtype=np.float64)
    for gi in range(gh):
        for gj in range(gw):
            cell_rng = np.random.default_rng(int(cell_seeds[gi, gj]))
            grid[gi, gj] = cell_rng.uniform(-1.0, 1.0)
```

```python
@dataclass
class WaterfallMistResult:
    """Output of the waterfall mist pass.

    mist_zone_mask : (H, W) float32, 0-1 intensity per cell.
    wet_surface_decal : list of dicts, each {"world_x": float, "world_y": float,
        "radius_m": float, "intensity": float}. Consumer: Unity wet-surface shader.
    """
    mist_zone_mask: np.ndarray
    wet_surface_decal: list


def pass_waterfall_mist(
    state: "TerrainPipelineState",
    region: "Optional[BBox]",
) -> "PassResult":
    """Bundle C supplementary pass: waterfall mist zone + wet-surface decal list.

    Runs AFTER pass_waterfalls (requires mist channel already populated).
    Produces:
        mist_zone_mask — (H, W) float32 copy of stack.mist (or zeros if absent)
        wet_surface_decal — written to stack as a Python list in stack channels dict
    """
    import time as _time
    t0 = _time.perf_counter()
    stack = state.mask_stack

    h_shape = stack.height.shape if stack.height is not None else (1, 1)

    # mist_zone_mask: copy of the mist channel produced by pass_waterfalls
    if stack.mist is not None:
        mist_zone_mask = np.asarray(stack.mist, dtype=np.float32).copy()
    else:
        mist_zone_mask = np.zeros(h_shape, dtype=np.float32)

    # wet_surface_decal: one entry per local maximum in the mist mask above threshold
    threshold = 0.3
    decal_list: list = []
    if np.any(mist_zone_mask > threshold):
        from scipy.ndimage import label as _label  # local import
        binary = mist_zone_mask > threshold
        labeled, num_features = _label(binary)
        cs = float(stack.cell_size) if stack.cell_size else 1.0
        ox = float(stack.world_origin_x) if stack.world_origin_x is not None else 0.0
        oy = float(stack.world_origin_y) if stack.world_origin_y is not None else 0.0
        for feat_id in range(1, num_features + 1):
            indices = np.argwhere(labeled == feat_id)
            if len(indices) == 0:
                continue
            centroid = indices.mean(axis=0)
            row_c, col_c = float(centroid[0]), float(centroid[1])
            intensity = float(mist_zone_mask[labeled == feat_id].max())
            radius_m = math.sqrt(len(indices)) * cs  # rough radius from area
            decal_list.append({
                "world_x": ox + col_c * cs,
                "world_y": oy + row_c * cs,
                "radius_m": float(radius_m),
                "intensity": intensity,
            })

    stack.set("mist_zone_mask", mist_zone_mask, "waterfall_mist")
    # wet_surface_decal is not an ndarray — store in a side channel dict
    stack._extra_channels = getattr(stack, "_extra_channels", {})
    stack._extra_channels["wet_surface_decal"] = decal_list

    return PassResult(
        pass_name="waterfall_mist",
        status="ok",
        duration_seconds=_time.perf_counter() - t0,
        consumed_channels=("mist",),
        produced_channels=("mist_zone_mask", "wet_surface_decal"),
        metrics={
            "mist_zone_cells": int(np.sum(mist_zone_mask > threshold)),
            "wet_surface_decal_count": len(decal_list),
        },
        issues=[],
    )


def register_bundle_c_mist_pass() -> None:
    """Register the Bundle C waterfall mist supplementary pass."""
    from .terrain_pipeline import TerrainPassController

    TerrainPassController.register_pass(
        PassDefinition(
            name="waterfall_mist",
            func=pass_waterfall_mist,

            seed_namespace="waterfall_mist",
            requires_scene_read=False,
            may_modify_geometry=False,
            description="Bundle C — waterfall mist zone mask + wet-surface decal list",
        )
    )
```

```python
def rasterize_poi_mask(
    stack: "TerrainMaskStack",
    pois: list,
    radius_m: float = 20.0,
) -> np.ndarray:
    """Rasterize a list of POI world-space positions into a (H, W) float32 mask.

    Each POI within ``radius_m`` metres contributes 1.0 at its centre,
    decaying linearly to 0.0 at the radius boundary. Overlapping POIs take
    the maximum of their contributions.

    Parameters
    ----------
    stack : TerrainMaskStack
        Must have ``height`` populated (for shape) and ``cell_size`` set.
    pois : list of (world_x, world_y) or (world_x, world_y, world_z) tuples
    radius_m : float
        Influence radius in world metres. Default 20.0.

    Side effect
    -----------
    Calls ``stack.set("poi_mask", mask, "rasterize_poi_mask")`` before returning.
    """
    from .terrain_semantics import TerrainMaskStack as _TMS  # local to avoid circular import

    if stack.height is None:
        raise ValueError("rasterize_poi_mask requires stack.height")

    h_arr = np.asarray(stack.height)
    H, W = h_arr.shape
    cs = float(stack.cell_size) if stack.cell_size else 1.0
    ox = float(stack.world_origin_x) if stack.world_origin_x is not None else 0.0
    oy = float(stack.world_origin_y) if stack.world_origin_y is not None else 0.0

    mask = np.zeros((H, W), dtype=np.float32)
    radius_cells = max(1, int(math.ceil(radius_m / cs)))

    # Row/col coordinate grids
    row_grid, col_grid = np.mgrid[0:H, 0:W].astype(np.float64)

    for poi in pois:
        wx = float(poi[0])
        wy = float(poi[1])
        # Convert world position to fractional cell indices
        col_c = (wx - ox) / cs - 0.5
        row_c = (wy - oy) / cs - 0.5

        # Only process cells within bounding box for performance
        r0 = max(0, int(row_c) - radius_cells - 1)
        r1 = min(H, int(row_c) + radius_cells + 2)
        c0 = max(0, int(col_c) - radius_cells - 1)
        c1 = min(W, int(col_c) + radius_cells + 2)

        dist = np.sqrt(
            ((row_grid[r0:r1, c0:c1] - row_c) * cs) ** 2
            + ((col_grid[r0:r1, c0:c1] - col_c) * cs) ** 2
        )
        contribution = np.clip(1.0 - dist / radius_m, 0.0, 1.0).astype(np.float32)
        mask[r0:r1, c0:c1] = np.maximum(mask[r0:r1, c0:c1], contribution)

    stack.set("poi_mask", mask, "rasterize_poi_mask")
    return mask
```
