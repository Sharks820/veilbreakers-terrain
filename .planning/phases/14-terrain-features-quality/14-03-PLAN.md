Replace with scipy EDT approach:
```python
from scipy.ndimage import distance_transform_edt

# Build a binary mask where path cells = 0 (foreground for EDT)
path_mask = np.ones((H, W), dtype=np.uint8)
for (pr, pc) in dense:
    ri = int(round(pr)); ci = int(round(pc))
    if 0 <= ri < H and 0 <= ci < W:
        path_mask[ri, ci] = 0

# EDT: distance in pixels from each cell to nearest foreground (0) cell
dist_pixels = distance_transform_edt(path_mask)  # (H, W) float64

# Crop to bounding box region only
dist_crop = dist_pixels[rmin:rmax, cmin:cmax]
frac = dist_crop / half_cells

# Apply U-valley profile vectorized
carve = np.where(frac < 0.3, 1.0,
        np.sqrt(np.maximum(0.0, 1.0 - ((frac - 0.3) / 0.7) ** 2)))
carve[dist_crop >= half_cells] = 0.0
delta[rmin:rmax, cmin:cmax] = -depth_m * carve
```

```python
def _compute_tile_contracts(
    tile_origin: tuple[float, float],
    tile_size_m: float,
    line_start: tuple[float, float],
    line_end: tuple[float, float],
) -> list[float]:
    """Return parametric t-values where the line segment crosses tile edges.

    Uses the parametric slab test (Smits' method):
        t = (boundary - start) / (end - start)
    for each of the 4 AABB edges. Returns all t in [0,1] where the segment
    intersects an edge of the tile AABB.

    Args:
        tile_origin: (x0, y0) bottom-left corner of tile.
        tile_size_m: Tile side length in meters.
        line_start: (sx, sy) segment start in world space.
        line_end: (ex, ey) segment end in world space.

    Returns:
        Sorted list of t values in [0, 1] at AABB edge crossings.
        Empty if the segment does not cross the tile.
    """
    x0, y0 = tile_origin
    x1, y1 = x0 + tile_size_m, y0 + tile_size_m
    sx, sy = line_start
    ex, ey = line_end

    dx = ex - sx
    dy = ey - sy

    ts: list[float] = []
    EPS = 1e-9

    # Slab test on X axis
    if abs(dx) > EPS:
        t_x0 = (x0 - sx) / dx
        t_x1 = (x1 - sx) / dx
        for t in (t_x0, t_x1):
            if 0.0 <= t <= 1.0:
                py = sy + t * dy
                if y0 <= py <= y1:
                    ts.append(t)

    # Slab test on Y axis
    if abs(dy) > EPS:
        t_y0 = (y0 - sy) / dy
        t_y1 = (y1 - sy) / dy
        for t in (t_y0, t_y1):
            if 0.0 <= t <= 1.0:
                px = sx + t * dx
                if x0 <= px <= x1:
                    ts.append(t)

    return sorted(set(round(t, 9) for t in ts))
```
